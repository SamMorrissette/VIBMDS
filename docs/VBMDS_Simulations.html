Simulate data

In each simulation, data is generated from a 6-component mixture distribution with mean vectors simulated uniformly on a 4-dimensional hypercube with sides [-1, 1], with mixture proportions drawn from a Dirichlet distribution.

gen_data <- function(n, p, sigma) {
  # Generate from a mixture: 
  mix_proportions <- rdirichlet(1, rep(2, G))
  mix_params <- MixSim(BarOmega = 0.01, K = G, p = p,
                    sph = FALSE, hom = FALSE, int = c(-1, 1))
  sim_data <- simdataset(n = n, Pi = mix_proportions, Mu = mix_params$Mu, S = mix_params$S)
  
  dist_mat <- dist(sim_data$X)
  d_unobs <- c(dist_mat) # Vector of dissimilarities
  
  m <- length(d_unobs)
  # Perturb the distances:
  d_obs <- rtruncnorm(m, a = 0, mean = d_unobs, sd = sigma) 
  
  d_obs_mat <- structure(d_obs, Size = n, class = "dist")
    
  list(sim_data = sim_data, 
       D = as.matrix(d_obs_mat))
}

Each element of the list will contain the true, unobserved, data and the observed dissimilarities.

G <- 6
data_list <- list()

Prior Parameters

For \(\lambda_k\), \(k = 1, \dots, K\), and \(\sigma\), we use a half-Cauchy prior with a large scale a parameter so that it is uninformative. We store the priors as a list of torch tensors.

prior_params <- list(scale_lambda = torch_tensor(25.0, device = device),
                     scale_sigma = torch_tensor(25.0, device = device))

Execution

For each method, we implement a function that:

  1. Executes the algorithm,

  2. Keeps track of the time required to complete the algorithm,

  3. Calculates Sum of Squared Error (SSE) between the fitted distances and the original (unperturbed) distances to evaluate the recovery of the true object configuration.

  4. Returns the fitted distances and the above metrics.

VBMDS:

run_vbmds <- function(data, p, prior_params, B, S, device) {
  # Run algorithm:
  begin <- Sys.time()
  vbmds_out <- vi_bmds(data$D, p, prior_params, B, S, max_iter = 5000, device = device)
  end <- Sys.time()
  time_elapsed <- as.numeric(difftime(end, begin, units = "secs"))

  
  # Generate unobserved distances using the optimized parameters:
  num_samples <- 1000
  q_samples <- generate_z(vbmds_out, num_samples)
  n <- dim(data$D)[1]
  m <- choose(n, 2)
  d_sum <- numeric(m)
  for (s in 1:num_samples) {
    d_sum <- d_sum + c(dist(as.matrix(q_samples$x[s,,])))
  }
  fitted_dists <- d_sum / num_samples
  
  # Calculate SSE: 
  sim_dists <- c(dist(data$sim_data$X))
  sse <- sum((fitted_dists - sim_dists)^2)
  
  list(time_elapsed = time_elapsed, 
       sse = sse)
}

BMDS:

run_bmds <- function(data, p) {
  # Run algorithm:
  begin <- Sys.time()
  bmds_out <-  bmds(data$D, min_p = p, max_p = p)
  end <- Sys.time()
  time_elapsed <- as.numeric(difftime(end, begin, units = "secs"))
  
  # Average over the MCMC samples to obtain mean distances:
  dist_samples <- bmds_out$del.L[[p]]
  num_samples <- nrow(dist_samples)
  m <- ncol(dist_samples)
  
  d_sum <- numeric(m)
  for (s in 1:num_samples) {
    d_sum <- d_sum + dist_samples[s,]
  }
  fitted_dists <- d_sum / num_samples
  
  # Calculate SSE: 
  sim_dists <- c(dist(data$sim_data$X))
  sse <- sum((fitted_dists - sim_dists)^2)
  
  list(time_elapsed = time_elapsed, 
       sse = sse)
}

MDS:

run_mds <- function(data, p) {
  begin <- Sys.time()
  mds_out <- cmdscale(data$D, k = p)
  end <- Sys.time()
  time_elapsed <- as.numeric(difftime(end, begin, units = "secs"))
  
  # Obtain distances: 
  fitted_dists <- c(dist(mds_out))
  
  # Calculate SSE:
  sim_dists <- c(dist(data$sim_data$X))
  sse <- sum(c(fitted_dists - sim_dists)^2)
  
  list(time_elapsed = time_elapsed, 
       sse = sse)
}

Simulation Loop:

After \(n > 500\), we will no longer be able to run BMDS due to memory and time limitations.

time_vbmds <- numeric(num_sim)
time_bmds <- rep(NA, num_sim)
time_mds <- numeric(num_sim)

sse_vbmds <- numeric(num_sim)
sse_bmds <- rep(NA, num_sim)
sse_mds <- numeric(num_sim)

for (sim in 1:num_sim) {
  data <- data_list[[sim]]
  
  vbmds_out <- run_vbmds(data, true_dim, prior_params, B[sim], S, device)
  time_vbmds[sim] <- vbmds_out$time_elapsed
  sse_vbmds[sim] <- vbmds_out$sse
  
  if (n[sim] <= 500) {
    bmds_out <- run_bmds(data, true_dim)
    time_bmds[sim] <- bmds_out$time_elapsed
    sse_bmds[sim] <- bmds_out$sse
  }
  
  mds_out <- run_mds(data, true_dim)
  time_mds[sim] <- mds_out$time_elapsed
  sse_mds[sim] <- mds_out$sse
}
